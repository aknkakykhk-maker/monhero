const path=require('path'),fs=require('fs'),http=require('http');
const ROOT='/home/user/monhero';
const { chromium } = require(path.join(ROOT,'tools','node_modules','playwright'));
const { quietBootSeed, updateNoticeSeed } = require(path.join(ROOT,'tools','boot','quiet-boot-seed'));
const OUT=process.argv[2]; fs.mkdirSync(OUT,{recursive:true});
const TYPES={'.html':'text/html','.js':'text/javascript','.css':'text/css','.png':'image/png','.jpg':'image/jpeg','.webp':'image/webp','.json':'application/json','.mp3':'audio/mpeg'};
(async()=>{
const server=await new Promise(r=>{const s=http.createServer((q,res)=>{const f=path.join(ROOT,decodeURIComponent(q.url.split('?')[0]));if(!f.startsWith(ROOT)||!fs.existsSync(f)||fs.statSync(f).isDirectory()){res.writeHead(404);res.end();return;}res.writeHead(200,{'Content-Type':TYPES[path.extname(f)]||'application/octet-stream'});fs.createReadStream(f).pipe(res);});s.listen(8982,()=>r(s));});

const errs=[];page.on('pageerror',e=>errs.push(String(e.message||e)));
await page.addInitScript(quietBootSeed());await page.addInitScript(updateNoticeSeed());
await page.goto('http://localhost:8982/monster-hero/index.html',{waitUntil:'load',timeout:90000});

   window.__run=run;
   const v=SUC.peek(run.s),ev=SUC.evaluate(run.s);
   return React.createElement('div',{className:'msx',style:{height:'100%',display:'flex',flexDirection:'column'}},React.createElement('style',null,MSX_CSS),
    which==='stats'?React.createElement(MsxStats,{run,setRun,v,ev,onClose(){}}):React.createElement(MsxR3Retire,{run,setRun,onPause(){},debug:false,SUC}));}
  ReactDOM.flushSync(()=>window.__r.render(React.createElement(W)));
 },{which});
 await page.waitForTimeout(600);
 const sel=which==='stats'?'[data-msx-pts-row="pow"] .sym.plus':'[data-msx-r3-plus="pow"]';

 const val=()=>page.evaluate(({which})=>{ if(which==='stats'){const r=document.querySelector('[data-msx-pts-row="pow"]');return r.innerText.replace(/\s+/g,' ');} return document.querySelector('.v-pa > div:nth-child(2)').innerText.replace(/\s+/g,' ');},{which});
 const out=[`== ${which}`,'初め: '+await val()];

  for(let i=0;i<5;i++) await page.click(minusSel,{timeout:500}).catch(()=>{});

 }
 // 長押し(1P): 1.2秒
