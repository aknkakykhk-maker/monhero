// 関数ごとの 自分の時間 / 呼んだ先を含む時間(ms)
function summarizeProfile(prof) {
  const byId = new Map(prof.nodes.map(n => [n.id, n]));
  const parent = new Map(); for (const n of prof.nodes) for (const c of (n.children || [])) parent.set(c, n.id);
  const dt = new Map(); // node id → ms
  for (let i = 0; i < prof.samples.length; i++) dt.set(prof.samples[i], (dt.get(prof.samples[i]) || 0) + (prof.timeDeltas[i + 1] || 0) / 1000);
  const key = (n) => { const c = n.callFrame; return `${c.functionName || '(無名)'} ${(c.url || '').split('/').pop().split('?')[0]}:${c.lineNumber + 1}`; };
  const self = {}, total = {};
  for (const [id, ms] of dt) {
    const n = byId.get(id); self[key(n)] = (self[key(n)] || 0) + ms;
    const seen = new Set(); let cur = id;
    while (cur != null) { const k = key(byId.get(cur)); if (!seen.has(k)) { total[k] = (total[k] || 0) + ms; seen.add(k); } cur = parent.get(cur); }
  }
  const top = (o, n) => Object.entries(o).sort((a, b) => b[1] - a[1]).slice(0, n).map(([k, v]) => [k, Math.round(v)]);
  const all = Object.values(self).reduce((a, b) => a + b, 0);
  return { allMs: Math.round(all), self: top(self, 60), total: top(total, 120) };
}
