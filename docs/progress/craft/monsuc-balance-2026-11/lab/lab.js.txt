const patchTo = (src, reps, name) => {
  let t = fs.readFileSync(src, 'utf8');
  for (const [re, by] of reps) { const before = t; t = t.replace(re, by); if (t === before) throw new Error(`置き換え失敗 ${name} ${re}`); }
  fs.mkdirSync(PATCH_DIR, { recursive: true });
  const out = path.join(PATCH_DIR, `${name}-${path.basename(src)}`); fs.writeFileSync(out, t); return out;
};
const useVariant = (v) => {
  const V = VARIANTS[v]; if (!V) throw new Error(`知らない仮の数値 ${v}`);
  const map = {};
  if (V.core) map[CORE] = patchTo(CORE, V.core, v);
  if (V.ev) map[EVF] = patchTo(EVF, V.ev, v);
  if (V.arc) map[ARC] = patchTo(ARC, V.arc, v);
  const orig = Module._resolveFilename;
  Module._resolveFilename = function (req, parent, ...rest) { const r = orig.call(this, req, parent, ...rest); return map[r] || r; };
};

const K = ['life', 'pow', 'wise', 'hit', 'eva', 'tough'];
const STRATS = ['train', 'shugyo', 'balance', 'tag', 'hp', 'cup', 'ev'];
const REACTS = ['praise', 'scold', 'smart', 'none']; // ほめるだけ / しかるだけ / 良ければほめ悪ければしかる / なにもしない
const REAL = { B: { sum: 2600, techs: 4, loy: 70, pop: 40, power: 10000 }, A: { sum: 3400, techs: 6, loy: 80, pop: 55, power: 12500 }, S: { sum: 4300, techs: 8, loy: 85, pop: 70, power: 14500 } };
const MAIN_TEAMS = ['base', 'box-mid', 'real-B', 'real-A', 'real-S', 'battle-S'];
const TEAM_SP = ['Undine', 'Mocchi', 'Golem', 'Pixie', 'Tiger', 'Kyubi'];
const h32 = (x) => { x = (x ^ 61) ^ (x >>> 16); x = (x + (x << 3)) >>> 0; x ^= x >>> 4; x = Math.imul(x, 0x27d4eb2d) >>> 0; x ^= x >>> 15; return x >>> 0; };

const makeLab = (variant = 'base') => {
  useVariant(variant);
  const { CORE: C2, loadDeps, depsFor } = require(path.join(ROOT, 'tools/success/load'));
  const battleLoad = require(path.join(ROOT, 'tools/masmon-battle/load.js'));
  const { makeGameFight } = require(path.join(ROOT, 'tools/success/game-fight.js'));
  const { boxSeed } = require(path.join(ROOT, 'tools/success/monsuc-play10.js'));
  const AT = require(path.join(ROOT, 'monster-hero/data/monhero-success-assist.js'));
  const SUC = require(C2).create(depsFor(loadDeps()));
  const { engine, catalog } = battleLoad.load();
  const GAME = makeGameFight(catalog);
  const species = SUC.speciesList.slice();
  const aids = SUC.AIDS.slice();

  const raisedTeam = (rank, picks, count = 6) => {
    const p = REAL[rank];
    return TEAM_SP.slice(0, count).map((x, i) => {
      const st = {}; K.forEach((k, j) => { st[k] = Math.round(p.sum / 6 * (0.7 + ((i + j) % 4) * 0.2)); });
      return { kind: 'raised', species: x, name: `箱${i + 1}`, stats: st, lv: 30, loy: p.loy, pop: p.pop, techs: SUC.techsOf(x).slice(0, p.techs).map((t) => t.id), assists: picks };
    });
  };
  const teamOf = (kind) => {
    if (kind === 'base') return [];
    if (kind.startsWith('box-')) return boxSeed(kind).mons.slice(0, 6).map((m) => ({ kind: 'raised', species: m.species, name: m.nick, stats: m.st, techs: m.techs }));
    let m;
    if ((m = kind.match(/^(real|battle)-([BAS])$/))) return raisedTeam(m[2], m[1] === 'battle' ? AT.LIST.filter((a) => a.type === 'battle').map((a) => ({ id: a.id, n: a.max })) : AT.autoPick(AT.pointsOf(REAL[m[2]].power)).picks);
    if ((m = kind.match(/^memo(\d)-([BAS])$/))) return raisedTeam(m[2], AT.autoPick(AT.pointsOf(REAL[m[2]].power)).picks, Number(m[1])); // 残りはベースモン
    if ((m = kind.match(/^bare-([BAS])$/))) return raisedTeam(m[1], []);                                                      // 能力は同じ・アシストなし
    if ((m = kind.match(/^probe-(\w+)$/))) return raisedTeam('S', [{ id: m[1], n: AT.LIST.find((a) => a.id === m[1]).max }]);  // 1種だけ最大
    if ((m = kind.match(/^half-(\w+)$/))) { const a = AT.LIST.find((x) => x.id === m[1]); return raisedTeam('S', [{ id: m[1], n: Math.max(1, Math.round(a.max / 2)) }]); }
    throw new Error(`知らない編成 ${kind}`);
  };

  // 条件: 主の回し方は run.js と同じ歩幅。反応(ほめ・しかり)は i の別の散らしで決める
  const combos = species.length * STRATS.length * MAIN_TEAMS.length * aids.length;
  const gcd = (a, b) => (b ? gcd(b, a % b) : a);
  let stride = Math.floor(combos * 0.618) | 1; while (gcd(stride, combos) !== 1) stride += 2;
  const condOf = (i, teams = MAIN_TEAMS) => {
    let n = (i % combos) * stride % combos;
    const sp = species[n % species.length]; n = Math.floor(n / species.length);
    const st = STRATS[n % STRATS.length]; n = Math.floor(n / STRATS.length);
    const tm = MAIN_TEAMS[n % MAIN_TEAMS.length]; n = Math.floor(n / MAIN_TEAMS.length);
    const aid = aids[n % aids.length];
    const react = REACTS[h32(i * 2 + 1) % REACTS.length];
    return { species: sp, strategy: st, team: teams === MAIN_TEAMS ? tm : null, aid, react };
  };

  const chooseAct = (s, vw, strategy, rnd, focus) => {
    if (vw.can.includes('cond')) return { kind: 'cond' };
    if (vw.can.includes('retry')) return { kind: 'retry' };
    if (vw.can.includes('clinic')) return { kind: 'clinic' };
    const hp = s.hpMax > 0 ? s.hp / s.hpMax : 1;
    const cards = K.map((k) => ({ stat: k, c: vw.cards && vw.cards[`train:${k}`] })).filter((x) => x.c);
    const ev = (x) => (x.c.mult || 1) * (1 - (x.c.fail || 0)) * (1 + (x.c.great || 0)) * (focus.includes(x.stat) ? 1.3 : 1);
    const best = (score) => cards.slice().sort((a, b) => score(b) - score(a) || K.indexOf(a.stat) - K.indexOf(b.stat))[0];
    const train = (score = ev) => { const b = best(score); return b ? { kind: 'train', stat: b.stat } : { kind: 'rest' }; };
    const shugyoOk = vw.can.includes('shugyo') && vw.shugyo && vw.shugyo.techs && vw.shugyo.techs.length && !vw.shugyo.short;
    const shugyo = () => { const t = vw.shugyo.techs.slice().sort((x, y) => (x.rank || 0) - (y.rank || 0)).pop(); return { kind: 'shugyo', course: 'fire', techs: [t.id] }; };
    const cup = (vw.cups || []).find((c) => c.entry && c.entry.ok);
    switch (strategy) {
      case 'train': return hp < 0.3 ? { kind: 'rest' } : train((x) => (focus.includes(x.stat) ? 2 : 1) * (x.c.mult || 1));
      case 'shugyo': if (hp < 0.3) return { kind: 'rest' }; return shugyoOk ? shugyo() : train();
      case 'tag': return hp < 0.3 ? { kind: 'rest' } : train((x) => (x.c.tags ? x.c.tags.length : 0) * 10 + (x.c.members ? x.c.members.length : 0) + (x.c.mult || 1));
      case 'hp': if (hp < 0.5) return { kind: 'rest' }; if (s.stress > 60) return { kind: 'play' }; return train();
      case 'cup': if (cup && hp >= 0.4) return { kind: 'cup', id: cup.id }; return hp < 0.5 ? { kind: 'rest' } : train();
      case 'ev': {
        if (hp < 0.35) return { kind: 'rest' };
        const b = best(ev);
        if (shugyoOk && (!b || ev(b) < 1.15)) return shugyo();
        return b ? { kind: 'train', stat: b.stat } : { kind: 'rest' };
      }
      default: {
        const w = { tr: 4, rest: 2, play: 1, shu: shugyoOk ? 2 : 0, cup: cup ? 1 : 0 };
        if (hp < 0.3) { w.rest += 8; w.tr = 0; w.shu = 0; } else if (hp < 0.5) w.rest += 3;
        if (s.stress > 70) w.play += 5;
        const ks = Object.keys(w).filter((k) => w[k] > 0); let t = ks.reduce((a, k) => a + w[k], 0) * rnd();
        let pick = ks[0]; for (const k of ks) { t -= w[k]; if (t <= 0) { pick = k; break; } }
        if (pick === 'tr') return { kind: 'train', stat: focus[Math.floor(rnd() * focus.length)] };
        if (pick === 'shu') return shugyo();
        if (pick === 'cup') return { kind: 'cup', id: cup.id };
        return { kind: pick === 'play' ? 'play' : 'rest' };
      }
    }
  };

  // 出来事の選び方: 体力が上限の7割未満で、体力が増える選択肢があればそれ。ほかは最初に選べる物(run.js と同じ)
  const EVDEF = SUC.events && SUC.events.BY_ID ? SUC.events.BY_ID : null;
  const choiceHp = (e, c) => { const d = EVDEF && EVDEF[e.id]; const ch = d && d.ch && (d.ch[c] || (!d.ch.a && d.ch)); const fx = ch ? (ch.fx || ch.ok || []) : []; return fx.filter((f) => f[0] === 'hp' && f[1] > 0).reduce((t, f) => t + f[1], 0); };

  const runOne = (i, { team = null, cond: cond0 = null } = {}) => {
    const cond = cond0 || condOf(i);
    if (team) cond.team = team;
    let s = SUC.newRun({ id: `lab-${i}`, species: cond.species, aid: cond.aid, team: teamOf(cond.team), aidBondLv: 1 });
    let r = ((i + 1) * 2654435761 + 7) >>> 0;
    const rnd = () => { r = (r * 1103515245 + 12345) & 0x7fffffff; return r / 0x7fffffff; };
    const focus = K.slice().sort((a, b) => (s.stats[b] - s.stats[a]) || K.indexOf(a) - K.indexOf(b)).slice(0, 3);
    const out = { matches: 0, wins: 0, league: [0, 0], summit: [0, 0], cup: [0, 0], acts: {} };
    const loy = {};   // 理由ごとの [甘え, 恐れ, 回数]
    const hpSrc = {}; // 体力の出入り: 理由ごとの [合計, 回数]
    const evHp = {};  // 体力の増える出来事: id ごとの [合計, 回数]
    const addLoy = () => { (s.lw || []).forEach((x) => { const o = (loy[x.w] = loy[x.w] || [0, 0, 0]); o[0] += x.a; o[1] += x.f; o[2] += 1; }); s.lw = []; };
    const addHp = (k, d) => { const o = (hpSrc[k] = hpSrc[k] || [0, 0]); o[0] += d; o[1] += 1; };
    const polTurns = [0, 0, 0, 0, 0, 0, 0]; let ama40 = null, fear40 = null, amaHiTurns = 0, fearHiTurns = 0, injuries = 0, ills = 0;
    const amaAt = {}; // 期ごとの甘え・恐れ(ターン 12・24・36・終わり)
    let guard = 0; s.lw = [];
    const step = (f, tag) => { const h0 = s.hp; s = f(); const d = s.hp - h0; if (d) addHp(tag, d); addLoy(); };
    while (!s.over && guard++ < 300) {
      if (s.battle) {
        const b = s.battle;
        const f = GAME.msxR3Fight(engine, SUC, { s, nick: '' });
        const won = Boolean(f && f.won);
        out.matches += 1; if (won) out.wins += 1;
        const slot = b.kind === 'summit' ? 'summit' : b.kind === 'cup' ? 'cup' : 'league';
        out[slot][1] += 1; if (won) out[slot][0] += 1;
        step(() => SUC.settle(s, [won], f ? f.use : null).state, 'battle');
        if (s.mealReady && !s.over) { const o = SUC.mealOpts(s); if (o[0]) step(() => SUC.meal(s, o[0]).state, 'meal'); }
        continue;
      }
      if (SUC.storyDue(s) !== null) step(() => SUC.storyStep(s, 'a').state, 'story');
      const ev = (slot) => {
        if (!SUC.events) return;
        const x = SUC.events.open(s, slot); s = x.state; const e = x.out.ev;
        if (!e) return;
        if (e.watch) { const h0 = s.hp; const y = SUC.events.pick(s, null); if (y.out.ok) { s = y.state; const d = s.hp - h0; if (d) { addHp(d > 0 ? 'event+' : 'event-', d); if (d > 0) { const o = (evHp[e.id] = evHp[e.id] || [0, 0]); o[0] += d; o[1] += 1; } } addLoy(); } return; }
        const ok = e.choices.filter((c) => c.ok); if (!ok.length) return;
        let pick = ok[0];
        if (s.hp < s.hpMax * 0.7) { const hpc = ok.filter((c) => choiceHp(e, c.c) > 0).sort((a, b) => choiceHp(e, b.c) - choiceHp(e, a.c))[0]; if (hpc) pick = hpc; }
        const h0 = s.hp;
        const y = SUC.events.pick(s, pick.c); if (y.out.ok) { s = y.state; const d = s.hp - h0; if (d) { addHp(d > 0 ? 'event+' : 'event-', d); if (d > 0) { const o = (evHp[e.id] = evHp[e.id] || [0, 0]); o[0] += d; o[1] += 1; } } addLoy(); }
      };
      ev('start');
      const a = chooseAct(s, SUC.peek(s), cond.strategy, rnd, focus);
      let kind = a.kind; const condBefore = s.cond;
      const h0 = s.hp;
      let x = SUC.act(s, a);
      if (!x.out.ok) { x = SUC.act(s, { kind: 'rest' }); kind = 'rest'; }
      s = x.state; { const d = s.hp - h0; if (d) addHp(`act:${kind}`, d); } addLoy();
      if (!condBefore && s.cond) { if (s.cond.kind === 'injury') injuries++; else if (s.cond.kind === 'ill') ills++; }
      out.acts[kind] = (out.acts[kind] || 0) + 1;
      if (s.last && ['train', 'shugyo'].includes(s.last.kind)) {
        const good = s.last.res === 'great' || s.last.res === 'ok';
        const how = cond.react === 'praise' ? 'praise' : cond.react === 'scold' ? 'scold' : cond.react === 'smart' ? (good ? 'praise' : 'scold') : null;
        step(() => SUC.react(s, how).state, 'react'); ev('after');
      }
      if (SUC.issaComes(s)) step(() => SUC.issa(s, 'stat').state, 'issa');
      step(() => SUC.endTurn(s).state, 'endTurn');
      polTurns[Math.max(-3, Math.min(3, s.pol)) + 3] += 1;
      if (s.ama >= 40) { amaHiTurns++; if (ama40 === null) ama40 = s.turn; }
      if (s.fear >= 40) { fearHiTurns++; if (fear40 === null) fear40 = s.turn; }
      if ([12, 24, 36].includes(s.turn)) amaAt[s.turn] = [s.ama, s.fear];
    }
    s = SUC.autoPts(s).state;
    const e = SUC.evaluate(s);
    const statSum = K.reduce((t, k) => t + (Number(s.stats[k]) || 0), 0);
    const L = s.log || {};
    const bonds = Array.isArray(s.bond) ? s.bond.filter(Number.isFinite) : [];
    return {
      i, ...cond, endTurn: Number(s.turn) || 0, power: e.power, grade: String((e.grade && e.grade.label) || e.grade).replace(/\d+$/, ''),
      statSum, stats: K.map((k) => Math.round(s.stats[k] || 0)), techs: Array.isArray(s.techs) ? s.techs.length : 0, ...out,
      ama: s.ama, fear: s.fear, pol: s.pol, polTurns, ama40, fear40, amaHiTurns, fearHiTurns, amaAt, loy,
      hpSrc, evHp, injuries, ills, hpMax: s.hpMax,
      log: { train: L.train, great: L.great, fail: L.fail, sab: L.sab, rest: L.rest, shugyo: L.shugyo, learned: L.learned, tag: (L.tagTurns || []).slice(1).reduce((a, b) => a + b, 0), issa: (L.issa || []).join(',') },
      bondAvg: bonds.length ? Math.round(bonds.reduce((a, b) => a + b, 0) / bonds.length) : null,
    };
  };
  return { SUC, AT, condOf, runOne, MAIN_TEAMS };
};
module.exports = { makeLab, VARIANTS, REACTS, STRATS };
